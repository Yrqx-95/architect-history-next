function reject(termJa: string, reason: string): CandidateReviewRecord {
  const candidate = getCandidate(termJa)
  if (!candidate) throw new Error(`Rejected term is not in candidate pool: ${termJa}`)
  return {
    termJa,
    normalizedTermJa: candidate.normalizedTermJa,
    category: inferCategory(candidate),
    qualityGrade: 'C',
    sourceRefs: candidate.sourceRefs,
    aliases: [],
    reason,
  }
}

export const rejectedCandidates: CandidateReviewRecord[] = [
  ...[...productOrBrandTerms].map(termJa => reject(
    termJa,
    'Brand, designer, or product-series name. Keep in a separately verified product registry rather than the general glossary.',
  )),
  ...[...properNameTerms].map(termJa => reject(
    termJa,
    'Proper name of a person, institution, place, or individual work. It belongs in archive entities, not the general glossary.',
  )),
  ...[...contextOnlyTerms].map(termJa => reject(
    termJa,
    'Context-specific phrase or assessment label with insufficient standalone glossary value.',
  )),
]

export const rejectedJapaneseTerms = new Set(rejectedCandidates.map(item => item.termJa))

const overlap = rejectedCandidates.filter(item => curatedJapaneseTerms.has(item.termJa))
if (overlap.length > 0) {
  throw new Error(`Accepted/rejected overlap: ${overlap.map(item => item.termJa).join(', ')}`)
}

if (rejectedCandidates.length !== rejectedJapaneseTerms.size) {
  throw new Error('Duplicate rejected candidate records')
}

export const unclassifiedAfterReject = candidatePool.filter(candidate =>
  !curatedJapaneseTerms.has(candidate.termJa) &&
  !rejectedJapaneseTerms.has(candidate.termJa)
)
