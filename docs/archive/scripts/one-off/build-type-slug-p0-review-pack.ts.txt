import fs from 'node:fs'
import path from 'node:path'
import { ROOT } from './supabase-script-utils'

type QueueItem = {
  priority: 'P0' | 'P1' | 'P2'
  building_slug: string
  display_name: string
  country_code: string | null
  city: string | null
  location: string | null
  year_start: number | null
  architect_slug: string | null
  current_style_slugs: string[]
  ai_tags: Record<string, unknown>
  description_summary: string
  significance_summary: string
  candidate_type_slug: string | null
  candidate_subtype: string | null
  confidence: 'medium' | 'reject_unsafe' | 'none'
  reason: string
  suggested_action: 'assign' | 'skip_unknown' | 'needs_research' | 'taxonomy_gap'
  review_source: 'sprint05_B' | 'sprint05_C' | 'manual_no_rule'
  review_notes: string[]
  score: number
}

type ManualReviewQueue = {
  generatedAt: string
  totalBuildings: number
  missingTypeSlug: number
  queue: QueueItem[]
}

type ReviewPackItem = QueueItem & {
  recommended_type_slug: string | null
  alternative_type_slug: string | null
  recommendation_reason: string
  evidence_fields_used: string[]
  confidence_for_human_review: 'high' | 'medium' | 'low'
  final_decision: ''
  decision_options: 'yes / no / choose alternative / needs external research'
}

const QUEUE_JSON = path.join(ROOT, 'reports/type-slug-manual-review-queue.json')
const PACK_JSON = path.join(ROOT, 'reports/type-slug-p0-review-pack.json')
const PACK_MD = path.join(ROOT, 'TYPE_SLUG_P0_REVIEW_PACK.md')

function readQueue() {
  if (!fs.existsSync(QUEUE_JSON)) {
    throw new Error(`Missing manual review queue: ${path.relative(ROOT, QUEUE_JSON)}`)
  }
  return JSON.parse(fs.readFileSync(QUEUE_JSON, 'utf8')) as ManualReviewQueue
}

function norm(value: string) {
  return value.normalize('NFKC').toLowerCase()
}

function has(text: string, pattern: RegExp) {
  return pattern.test(text)
}

function evidenceFields(item: QueueItem, reasonFields: string[]) {
  const fields = new Set<string>(['display_name', 'candidate_type_slug', 'candidate_subtype'])
  if (item.country_code) fields.add('country_code')
  if (item.year_start) fields.add('year_start')
  if (item.architect_slug) fields.add('architect_slug')
  if (item.current_style_slugs.length) fields.add('style_slugs')
  if (item.description_summary) fields.add('description_summary')
  if (item.significance_summary) fields.add('significance_summary')
  for (const field of reasonFields) fields.add(field)
  return [...fields]
}

function recommend(item: QueueItem): Pick<
  ReviewPackItem,
  | 'recommended_type_slug'
  | 'alternative_type_slug'
  | 'recommendation_reason'
  | 'evidence_fields_used'
  | 'confidence_for_human_review'
> {
  const name = norm(item.display_name)
  const candidate = item.candidate_type_slug

  if (has(name, /\bwater tower\b|bridge\b|footbridge\b/)) {
    return {
      recommended_type_slug: 'infrastructure',
      alternative_type_slug: candidate && candidate !== 'infrastructure' ? candidate : null,
      recommendation_reason: 'Name indicates infrastructure rather than a generic office or civic-public use.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'high',
    }
  }

  if (has(name, /\b(world trade center|trade center|headquarters|head office|office|corporate|bank|banca|lloyd's|tower|towers|tour\b|torres|hochhaus|skyscraper|leadenhall|wood street|madison avenue|sloane street|raffles place|one57|parc1|qv\.?1|fcg building|lippo centre|mlc centre)\b/)) {
    return {
      recommended_type_slug: 'office',
      alternative_type_slug: has(name, /\bbank\b|banca|lloyd's/) ? 'commercial' : null,
      recommendation_reason:
        'Name contains a strong office/commercial building signal such as tower, headquarters, bank, or named office building.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'high',
    }
  }

  if (has(name, /\bhotel\b/)) {
    return {
      recommended_type_slug: 'commercial',
      alternative_type_slug: candidate && candidate !== 'commercial' ? candidate : null,
      recommendation_reason: 'Hotel use is more consistent with commercial than sports/residential in the current taxonomy.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'high',
    }
  }

  if (has(name, /\bmuseum\b|gallery\b|arts centre|arts center|performing arts|wexner center for the arts|architecture and design|\bcultural\b|heritage center\b/)) {
    return {
      recommended_type_slug: 'cultural',
      alternative_type_slug: candidate && candidate !== 'cultural' ? candidate : null,
      recommendation_reason: 'Name contains cultural institution terms such as museum, gallery, arts, heritage, or design center.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'high',
    }
  }

  if (has(name, /\bsynagogue\b|friedhof\b|cemetery\b/)) {
    return {
      recommended_type_slug: 'religious',
      alternative_type_slug: has(name, /\bfriedhof\b|cemetery/) ? 'public-space' : null,
      recommendation_reason: 'Name indicates religious or burial-related use; cemetery-like cases should be checked manually.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: has(name, /\bfriedhof\b|cemetery/) ? 'medium' : 'high',
    }
  }

  if (has(name, /\bgymnasium\b|sports\b|stadium\b|palacio de los deportes\b/)) {
    return {
      recommended_type_slug: 'sports',
      alternative_type_slug: candidate && candidate !== 'sports' ? candidate : null,
      recommendation_reason: 'Name contains a sports venue signal.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'high',
    }
  }

  if (has(name, /\bmesse\b|conference center|conference centre|convention centre|convention center\b/)) {
    return {
      recommended_type_slug: 'civic-public',
      alternative_type_slug: 'cultural',
      recommendation_reason:
        'Convention/exhibition facilities fit civic-public better than transportation; cultural may be chosen if the venue is primarily arts/event oriented.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'medium',
    }
  }

  if (has(name, /\buniversity\b|college\b|campus\b|education\b|dormitory\b/)) {
    return {
      recommended_type_slug: 'educational',
      alternative_type_slug: has(name, /\bdormitory\b/) ? 'residential' : null,
      recommendation_reason:
        'Name indicates university, college, campus, or education use; dormitory cases may be residential depending on taxonomy choice.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'medium',
    }
  }

  if (has(name, /\bparc\b|park\b|parque\b|plaza\b|piazza\b|villette\b/)) {
    return {
      recommended_type_slug: 'public-space',
      alternative_type_slug: candidate && candidate !== 'public-space' ? candidate : null,
      recommendation_reason: 'Name indicates park/plaza/public urban space, but mixed-use names should be checked.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'medium',
    }
  }

  if (has(name, /\bhouse\b|residence\b|residences\b|landhaus\b|estate\b|studio\b|can lis\b|terassitalo\b|ypenhof\b/)) {
    return {
      recommended_type_slug: 'residential',
      alternative_type_slug: has(name, /\bstudio\b/) ? 'office' : null,
      recommendation_reason:
        'Name contains residential terms, but studio/estate cases may need confirmation of actual use.',
      evidence_fields_used: evidenceFields(item, ['display_name']),
      confidence_for_human_review: 'medium',
    }
  }

  if (has(name, /\bhall\b|forum\b|foundation\b|palace\b|palazzo\b|paleis\b|club\b|community complex\b|centre\b|center\b/)) {
    return {
      recommended_type_slug: candidate || 'civic-public',
      alternative_type_slug: candidate === 'civic-public' ? 'cultural' : 'civic-public',
      recommendation_reason:
        'Name has a civic/institutional venue term, but these words are broad and require human confirmation.',
      evidence_fields_used: evidenceFields(item, ['display_name', 'candidate_type_slug']),
      confidence_for_human_review: 'medium',
    }
  }

  return {
    recommended_type_slug: candidate,
    alternative_type_slug: null,
    recommendation_reason: 'No stronger pattern was found beyond the Sprint 07 candidate; external/context review is needed.',
    evidence_fields_used: evidenceFields(item, ['candidate_type_slug']),
    confidence_for_human_review: 'low',
  }
}

function countBy<T extends string | null>(items: ReviewPackItem[], getter: (item: ReviewPackItem) => T) {
  const counts = new Map<string, number>()
  for (const item of items) {
    const key = getter(item) || 'none'
    counts.set(key, (counts.get(key) || 0) + 1)
  }
  return [...counts.entries()]
    .map(([key, count]) => ({ key, count }))
    .sort((a, b) => b.count - a.count || a.key.localeCompare(b.key))
}

function priorityScore(item: ReviewPackItem) {
  const confidence = { high: 3, medium: 2, low: 1 }[item.confidence_for_human_review]
  const completeness = [item.country_code, item.year_start, item.architect_slug].filter(Boolean).length
  const ambiguityPenalty = item.alternative_type_slug ? 0 : 1
  return confidence * 10 + completeness * 2 + item.score + ambiguityPenalty
}

function esc(value: unknown) {
  return String(value ?? '').replaceAll('|', '\\|').replace(/\s+/g, ' ').trim()
}

function markdownReport(report: {
  generatedAt: string
  sourceQueueGeneratedAt: string
  p0Count: number
  confidenceCounts: Array<{ key: string; count: number }>
  recommendedDistribution: Array<{ key: string; count: number }>
  top30: ReviewPackItem[]
  items: ReviewPackItem[]
}) {
  const lines = [
    '# Type Slug P0 Review Pack',
    '',
    `Generated: ${report.generatedAt}`,
    `Source queue generated: ${report.sourceQueueGeneratedAt}`,
    '',
    '## Scope',
    '',
    '- No database writes were performed.',
    '- No migration was generated.',
    '- Only Sprint 07 P0 records are included.',
    '- `final_decision` is intentionally empty for human review.',
    '- Decision options: yes / no / choose alternative / needs external research.',
    '',
    '## Summary',
    '',
    `- P0 records: ${report.p0Count}`,
    '',
    '### Confidence For Human Review',
    '',
    '| Confidence | Count |',
    '|---|---:|',
  ]

  for (const confidence of ['high', 'medium', 'low']) {
    const found = report.confidenceCounts.find(item => item.key === confidence)
    lines.push(`| ${confidence} | ${found?.count || 0} |`)
  }

  lines.push('', '### Recommended type_slug Distribution', '', '| recommended_type_slug | Count |', '|---|---:|')
  for (const item of report.recommendedDistribution) lines.push(`| ${item.key} | ${item.count} |`)

  lines.push('', '## Top 30 For First Human Pass', '')
  lines.push('| # | Building | Name | Recommended | Alternative | Confidence | Reason |')
  lines.push('|---:|---|---|---|---|---|---|')
  report.top30.forEach((item, index) => {
    lines.push(
      `| ${index + 1} | ${item.building_slug} | ${esc(item.display_name)} | ${item.recommended_type_slug || ''} | ${item.alternative_type_slug || ''} | ${item.confidence_for_human_review} | ${esc(item.recommendation_reason)} |`
    )
  })

  lines.push('', '## Full P0 Review Pack', '')
  lines.push(
    '| Building | Name | Country | Year | Architect | Current Candidate | Recommended | Alternative | Confidence | Evidence Fields | Reason | Final Decision |'
  )
  lines.push('|---|---|---|---:|---|---|---|---|---|---|---|---|')
  for (const item of report.items) {
    lines.push(
      `| ${item.building_slug} | ${esc(item.display_name)} | ${item.country_code || ''} | ${item.year_start ?? ''} | ${item.architect_slug || ''} | ${item.candidate_type_slug || ''} | ${item.recommended_type_slug || ''} | ${item.alternative_type_slug || ''} | ${item.confidence_for_human_review} | ${item.evidence_fields_used.join(', ')} | ${esc(item.recommendation_reason)} | ${item.final_decision} |`
    )
  }

  lines.push('', '## JSON Artifact', '')
  lines.push('- `reports/type-slug-p0-review-pack.json`')
  lines.push('')
  return lines.join('\n')
}

function main() {
  const queue = readQueue()
  const p0 = queue.queue.filter(item => item.priority === 'P0')
  const items: ReviewPackItem[] = p0.map(item => {
    const recommendation = recommend(item)
    return {
      ...item,
      ...recommendation,
      final_decision: '',
      decision_options: 'yes / no / choose alternative / needs external research',
    }
  })

  items.sort((a, b) => {
    const confidenceOrder = { high: 0, medium: 1, low: 2 }
    return (
      confidenceOrder[a.confidence_for_human_review] - confidenceOrder[b.confidence_for_human_review] ||
      (a.recommended_type_slug || 'zz').localeCompare(b.recommended_type_slug || 'zz') ||
      a.building_slug.localeCompare(b.building_slug)
    )
  })

  const top30 = [...items].sort((a, b) => priorityScore(b) - priorityScore(a) || a.building_slug.localeCompare(b.building_slug)).slice(0, 30)
  const report = {
    generatedAt: new Date().toISOString(),
    sourceQueueGeneratedAt: queue.generatedAt,
    p0Count: items.length,
    confidenceCounts: countBy(items, item => item.confidence_for_human_review),
    recommendedDistribution: countBy(items, item => item.recommended_type_slug),
    top30,
    items,
  }

  fs.writeFileSync(PACK_JSON, `${JSON.stringify(report, null, 2)}\n`)
  fs.writeFileSync(PACK_MD, markdownReport(report))

  console.log('Generated P0 type_slug review pack')
  console.log(`- P0 records: ${report.p0Count}`)
  for (const confidence of ['high', 'medium', 'low']) {
    const found = report.confidenceCounts.find(item => item.key === confidence)
    console.log(`- ${confidence}: ${found?.count || 0}`)
  }
  console.log(`- report: ${path.relative(ROOT, PACK_MD)}`)
  console.log(`- json: ${path.relative(ROOT, PACK_JSON)}`)
}

main()
